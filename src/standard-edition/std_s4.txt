function newLine(t,s,q,loc){
  if(t.type==='RMA')return {sku:s,demand:q,toStock:0,toClean:0,toScrap:0};
  if(t.type==='SCRAP')return {sku:s,loc:loc||'WH/Stock',demand:q,done:q,reason:''};
  if(t.type==='RTV')return {sku:s,loc:loc||'Virtual/Scrap',demand:q,done:q};
  if(t.type==='IN'||t.type==='CLN_IN')return {sku:s,demand:q,done:q};
  return {sku:s,demand:q,done:Math.min(q,Math.max(0,qty(s,t.src)))};
}

function adjAdd(t,m,loc,onlyStock){
  D.skus.forEach(function(k){if(k.m!==m)return;if(onlyStock&&qty(k.id,loc)<=0)return;if(t.lines.some(function(l){return l.sku===k.id&&l.loc===loc}))return;t.lines.push({sku:k.id,loc:loc,onhand:qty(k.id,loc),counted:''})});
  t.lines.sort(function(a,b){return skuSort(a.sku,b.sku)});
}
