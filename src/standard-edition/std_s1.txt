/* Validate: turn a transfer into stock moves between locations */
function buildT(t){
  var mv=[],err=[],add=function(s,q,src,dst,n){if(q>0)mv.push({s:s,q:q,src:src,dst:dst,n:n||''})};
  t.lines.forEach(function(l){
    var d=+l.done||0;
    if(t.type==='IN')add(l.sku,d,'Partners/Vendors',t.dst||'WH/Stock');
    else if(t.type==='OUT')add(l.sku,d,'WH/Stock',t.dst);
    else if(t.type==='CLN_OUT')add(l.sku,d,'WH/Cleaning','Transit/Factory','Cleaning');
    else if(t.type==='CLN_IN')add(l.sku,d,'Transit/Factory','WH/Stock','Cleaned');
    else if(t.type==='INT')add(l.sku,d,t.src,t.dst);
    else if(t.type==='SCRAP'){add(l.sku,d,l.loc||'WH/Stock','Virtual/Scrap',l.reason||'');if(d>0&&!l.reason)err.push(variantPlain(l.sku)+': Scrap Reason is required.')}
    else if(t.type==='RTV')add(l.sku,d,l.loc||'Virtual/Scrap','Partners/Vendors');
    else if(t.type==='RMA'){
      var a=+l.toStock||0,b=+l.toClean||0,c=+l.toScrap||0,rec=a+b+c,miss=Math.max(0,(+l.demand||0)-rec);
      add(l.sku,a,t.src,'WH/Stock','Return to stock');add(l.sku,b,t.src,'WH/Cleaning','Rework (cleaning)');add(l.sku,c,t.src,'Virtual/Scrap','Scrap at inspection');
      add(l.sku,miss,t.src,'Virtual/Inventory loss','Missing in return '+t.ref);
    }
    else if(t.type==='ADJ'){
      if(l.counted===''||l.counted==null)return;
      var cur=qty(l.sku,l.loc),diff=(+l.counted)-cur;
      if(diff<0)add(l.sku,-diff,l.loc,'Virtual/Inventory loss',(t.reason||'Count')+' (shortage)');
      else if(diff>0)add(l.sku,diff,'Virtual/Inventory adjustment',l.loc,(t.reason||'Count')+' (surplus)');
    }
    else if(t.type==='LOSS'){
      var q=+l.done||0;
      if(l.res==='stock')add(l.sku,q,'Virtual/Inventory loss','WH/Stock','Found');
      else if(l.res==='clean')add(l.sku,q,'Virtual/Inventory loss','WH/Cleaning','Found (needs cleaning)');
      else add(l.sku,q,'Virtual/Inventory loss','Virtual/Write-off',l.amt?'Compensated '+fmt(l.amt)+' EGP':'Written off');
    }
  });
  if(!mv.length)err.push(t.type==='ADJ'?'Enter at least one Counted quantity.':'There is nothing to process: set a quantity on at least one line.');
  if((t.type==='IN'||(t.type==='INT'&&(t.dst==='SHOP/Stock'||t.src==='SHOP/Stock')))&&!t.attach)err.push('Attach the signed delivery note (required for this operation type).');
  if(t.type==='INT'&&!ROUTES[t.src+'>'+t.dst])err.push('Route '+t.src+' to '+t.dst+' is not allowed. Items needing cleaning must go through Send for Cleaning.');
  if(t.type==='ADJ'){var cf=conflicts(t);if(cf.length&&!t.ack)err.push('Stock changed after you opened this count ('+cf.length+' lines). Tick the confirmation box to apply anyway.')}
  return {mv:mv,err:err};
}

}

}
