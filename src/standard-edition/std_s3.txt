/* ================= Demand context (ABC, velocity, orders) ================= */
function DM(){
  if(DMC)return DMC;
  var hist=[],hm={},hs={},hc={},hg={},days={},pcs=0,vel={},types={};
  S.docs.forEach(function(t){
    if(t.type!=='OUT'||t.state==='cancel'||!t.meta||!t.meta.orders)return;
    t.meta.orders.forEach(function(o){
      if(t.meta.cx&&t.meta.cx[o.ref])return;
      hist.push({date:t.date,o:o,t:t});days[t.date]=1;
      o.lines.forEach(function(s){var k=SKU.get(s);hm[k.m]=(hm[k.m]||0)+1;hs[k.s]=(hs[k.s]||0)+1;hc[k.c]=(hc[k.c]||0)+1;pcs++});
      if(o.gov)hg[o.gov]=(hg[o.gov]||0)+1;types[o.type]=(types[o.type]||0)+1;
    });
  });
  var dl=Object.keys(days).sort(),from=addDays(UI.today,-(S.settings.lookback-1)),span=dl.length?Math.max(1,Math.min(S.settings.lookback,dayDiff(dl[0],UI.today)+1)):1;
  hist.forEach(function(x){if(x.date>=from&&x.date<=UI.today)x.o.lines.forEach(function(s){vel[s]=(vel[s]||0)+1/span})});
  var abc={},tot=0,cum=0;Object.keys(hm).forEach(function(m){tot+=hm[m]});
  Object.keys(hm).sort(function(a,b){return hm[b]-hm[a]}).forEach(function(m){abc[m]=cum/(tot||1)<.8?'A':cum/(tot||1)<.95?'B':'C';cum+=hm[m]});
  return (DMC={hist:hist,hm:hm,hs:hs,hc:hc,hg:hg,pcs:pcs,abc:abc,dl:dl,span:span,vel:vel,types:types,tot:tot});
}

function detectKits(){
  var dm=DM(),sig={};
  dm.hist.forEach(function(x){var comp={};x.o.lines.forEach(function(s){var k=SKU.get(s),key=k.m+'|'+k.c;comp[key]=(comp[key]||0)+1});var ks=Object.keys(comp).sort();if(ks.length<3)return;

    o.n++;if(x.date===UI.today){var zs={};x.o.lines.forEach(function(s){zs[SKU.get(s).s]=1});var zk=Object.keys(zs);if(zk.length===1)o.today[zk[0]]=(o.today[zk[0]]||0)+1;else o.mixed++}});
  var ks=Object.keys(sig).map(function(k){return sig[k]}).filter(function(b){return b.n>=3}).sort(function(a,b){return b.n-a.n});
  ks.forEach(function(b){b.cap={};SZ.forEach(function(z){var min=Infinity,lim=[];Object.keys(b.comp).forEach(function(k){var p=k.split('|'),id=IDX[p[0]+'|'+p[1]+'|'+z],have=id?Math.floor(qty(id,'WH/Stock')/b.comp[k]):0;if(have<min){min=have;lim=[k]}else if(have===min)lim.push(k)});b.cap[z]={n:min===Infinity?0:min,lim:lim}})});
  return ks;
}

var LINE_RE=/^(.*?) \( (.*?) \)-(.*)-(.*)-(\d+)$/,COLS={ref:'كود الاوردر العشوائي',wb:'بوليصة الشحن',ty:'حالة الاوردر الجزئية',gov:'المحافظة',total:'المبلغ توتال بالشحن للاوردر',items:'المنتجات المطلوبة في الاوردر'};
function aliasOf(code){return S.alias[code]||D.alias[code]||(/^FC-/.test(code)&&MODEL.has(code.slice(3))?code.slice(3):null)}
function importRows(rows,fname,partner){
  var H=rows[0].map(function(x){return String(x).trim()}),ix={};for(var k in COLS)ix[k]=H.indexOf(COLS[k]);
  if(ix.ref<0||ix.items<0)throw new Error('Required columns not found. The sheet must have the same layout as the Vendoor export.');
  var have={};S.docs.forEach(function(t){if(t.type==='OUT'&&t.meta&&t.meta.orders)t.meta.orders.forEach(function(o){have[t.partner+'|'+o.ref]=t.ref})});
  var orders=[],dups=[],unk={};
  rows.slice(1).forEach(function(r){
    var ref=String(r[ix.ref]||'').trim();if(!ref)return;
    if(have[partner+'|'+ref]){dups.push(ref);return}
    var o={ref:ref,wb:r[ix.wb],type:String(r[ix.ty]).indexOf('استبدال')>=0?'Exchange':'New',gov:r[ix.gov],total:r[ix.total],lines:[],bad:[]};
    String(r[ix.items]).split(', ').forEach(function(it){
      var m=LINE_RE.exec(it.trim());if(!m){o.bad.push([it,'','',1]);return}
      var code=m[1],size=m[3].trim(),color=m[4].trim(),q=+m[5],mc=aliasOf(code),id=mc&&IDX[mc+'|'+color+'|'+size];
      if(id){for(var i=0;i<q;i++)o.lines.push(id)}else{o.bad.push([code,size,color,q]);if(!mc)unk[code]=1}
    });
    orders.push(o);
  });
  return {orders:orders,dups:dups,unk:Object.keys(unk)};
}
function resolveBad(res){res.orders.forEach(function(o){var keep=[];o.bad.forEach(function(b){var mc=aliasOf(b[0]),id=mc&&IDX[mc+'|'+b[2]+'|'+b[1]];if(id){for(var i=0;i<b[3];i++)o.lines.push(id)}else keep.push(b)});o.bad=keep});res.unk=res.unk.filter(function(c){return !aliasOf(c)})}
