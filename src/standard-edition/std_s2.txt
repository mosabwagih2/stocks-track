function statusbar(t){
  var cur=stateOf(t),steps=['Draft','Waiting','Ready','Done'];
  if(cur==='Cancelled')steps=['Draft','Cancelled'];
  else if(t.state!=='draft'&&['IN','RMA','CLN_IN','ADJ','LOSS'].indexOf(t.type)>=0)steps=['Draft','Ready','Done'];
  var ix=steps.indexOf(cur);
  return '<ol class="sbar">'+steps.map(function(s,i){return '<li class="'+(i===ix?'on':i<ix?'past':'')+(s==='Cancelled'?' cx':'')+'" title="'+esc(ST_AR[s])+'">'+s+'</li>'}).join('')+'</ol>';
}

/* ================= Overview (operation type cards) ================= */
function otCounts(type){
  var o={ready:0,waiting:0,late:0,bo:0,draft:0};
  S.docs.forEach(function(t){if(t.type!==type)return;var st=stateOf(t);if(t.state==='draft')o.draft++;if(t.state==='open'){if(st==='Ready')o.ready++;if(st==='Waiting')o.waiting++;if(isLate(t))o.late++;if(t.backorderOf)o.bo++}});
  return o;
}
function kcard(type){
  var ot=OT[type],c=otCounts(type),main=c.ready?c.ready+' To Process':c.draft?c.draft+' Draft':'No open transfers';
  var links=[];
  if(c.waiting)links.push('<a data-a="go" data-r="transfers" data-t="'+type+'" data-f="Waiting">'+c.waiting+' Waiting</a>');
  if(c.late)links.push('<a data-a="go" data-r="transfers" data-t="'+type+'" data-f="Late">'+c.late+' Late</a>');
  if(c.bo)links.push('<a data-a="go" data-r="transfers" data-t="'+type+'" data-f="Back Orders">'+c.bo+' Back Orders</a>');
  if(c.draft&&c.ready)links.push('<a data-a="go" data-r="transfers" data-t="'+type+'" data-f="Draft">'+c.draft+' Draft</a>');
  return '<div class="kcard"><div class="kh"><div><b>'+esc(ot.name)+'</b><div class="g2" dir="rtl">'+esc(ot.ar)+'</div></div><span class="mono small">'+esc(ot.prefix.replace(/\/$/,''))+'</span></div>'+
    '<button class="btn '+(c.ready?'primary':'')+'" data-a="go" data-r="transfers" data-t="'+type+'" data-f="'+(c.ready?'Ready':c.draft?'Draft':'')+'">'+main+'</button>'+
    '<div class="klinks">'+links.join('')+'</div><div class="kf"><button class="btn sm ghost" data-a="newt" data-t="'+type+'">+ New</button>'+(type==='OUT'?'<button class="btn sm ghost" data-a="drawer" data-k="import">Import</button>':'')+'</div></div>';
}

function addDays(d,n){var p=String(d).split('-');return new Date(Date.UTC(+p[0],+p[1]-1,+p[2])+n*864e5).toISOString().slice(0,10)}
